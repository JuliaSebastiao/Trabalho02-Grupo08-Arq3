# Valores decimais e negativos em operacoes de ponto flutuante.

CONFIG ROB_ENTRIES 8
CONFIG ADD_RS 2
CONFIG MULT_RS 2
CONFIG ADD_LATENCY 1
CONFIG SUB_LATENCY 1
CONFIG MUL_LATENCY 1
CONFIG DIV_LATENCY 1

F2 = -7.5
F4 = 2.5
F6 = -3

ADD.D F8, F2, F4
MUL.D F10, F8, F6
DIV.D F12, F10, F4
SUB.D F14, F12, F2

EXPECT F8 = -5
EXPECT F10 = 15
EXPECT F12 = 6
EXPECT F14 = 13.5
EXPECT EXECUTE 4 = 8
EXPECT COMMIT 4 = 10
EXPECT CYCLES = 10
