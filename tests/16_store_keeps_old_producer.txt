# Store entre duas escritas em F2.
# O store deve manter a tag da primeira escrita, mesmo que uma escrita mais nova de F2 apareca depois.

CONFIG ROB_ENTRIES 8
CONFIG STORE_BUFFERS 1
CONFIG ADD_RS 2
CONFIG ADD_UNITS 2
CONFIG ADD_LATENCY 3
CONFIG SUB_LATENCY 1
CONFIG STORE_LATENCY 1

R1 = 100
F4 = 10
F6 = 1
F8 = 50
F10 = 8

ADD.D F2, F4, F6
S.D F2, 0(R1)
SUB.D F2, F8, F10

EXPECT F2 = 42
EXPECT MEM[100] = 11
EXPECT EXECUTE 2 = 6
EXPECT WRITE 3 = 6
EXPECT COMMIT 2 = 8
EXPECT COMMIT 3 = 9
EXPECT CYCLES = 9
