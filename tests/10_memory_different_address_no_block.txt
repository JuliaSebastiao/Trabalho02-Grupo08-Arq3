# Store e load em enderecos diferentes:
# O load pode avancar porque nao existe conflito de endereco com o store mais antigo.

CONFIG LOAD_BUFFERS 2
CONFIG STORE_BUFFERS 2
CONFIG LOAD_LATENCY 1
CONFIG STORE_LATENCY 4

R1 = 100
F2 = 55

MEM[104] = 9
