# Store apos duas escritas em F2.
# O store deve gravar a versao mais nova de F2, produzida pela SUB.D.

CONFIG ROB_ENTRIES 8
CONFIG STORE_BUFFERS 1
CONFIG ADD_RS 2
CONFIG ADD_UNITS 2
CONFIG ADD_LATENCY 3
CONFIG SUB_LATENCY 1
CONFIG STORE_LATENCY 1

R1 = 100
F4 = 10
F6 = 1
F8 = 50
F10 = 8

ADD.D F2, F4, F6
SUB.D F2, F8, F10
S.D F2, 0(R1)

EXPECT F2 = 42
EXPECT MEM[100] = 42
EXPECT WRITE 2 = 4
EXPECT EXECUTE 3 = 5
EXPECT COMMIT 3 = 8
EXPECT CYCLES = 8
