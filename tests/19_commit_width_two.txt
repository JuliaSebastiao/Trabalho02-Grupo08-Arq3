# Commit multiplo.
# Duas instrucoes independentes podem confirmar no mesmo ciclo quando COMMIT_WIDTH = 2.

CONFIG ROB_ENTRIES 6
CONFIG ADD_RS 2
CONFIG ISSUE_WIDTH 2
CONFIG COMMIT_WIDTH 2
CONFIG CDB_WIDTH 2
CONFIG ADD_UNITS 2
CONFIG ADD_LATENCY 1

F1 = 1
F2 = 2
F3 = 3
F4 = 4

ADD.D F10, F1, F2
ADD.D F12, F3, F4

EXPECT F10 = 3
EXPECT F12 = 7
EXPECT ISSUE 1 = 1
EXPECT ISSUE 2 = 1
EXPECT WRITE 1 = 3
EXPECT WRITE 2 = 3
EXPECT COMMIT 1 = 4
EXPECT COMMIT 2 = 4
EXPECT CYCLES = 4
